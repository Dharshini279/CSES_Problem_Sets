n=int(input())
arr=[i for i in range(1,n+1)]
eve=[i for i in arr if i%2==0]
odd=[i for i in arr if i%2!=0]
if n>4:
    for i in range(len(eve)-1,-1,-1):
        print(eve[i],end=" ")
    for i in range(len(odd)-1,-1,-1):
        print(odd[i],end=" ")
elif n==1:
    print(1)
elif n==4:
    for i in range(len(eve)):
        print(eve[i],end=" ")
    for i in range(len(odd)):
        print(odd[i],end=" ")
else:
    print("NO SOLUTION")
